------------------------------------
  // INotifyPropertyChanged 实现 — 所有 ViewModel 的基类
  public class ViewModelBase : INotifyPropertyChanged
  {
      public event PropertyChangedEventHandler PropertyChanged;
      protected void OnPropertyChanged([CallerMemberName] string p = null)
          => PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(p));
      protected bool SetProperty<T>(ref T field, T value, [CallerMemberName] string p = null)
      {
          if (Equals(field, value)) return false;
          field = value; OnPropertyChanged(p); return true;
      }
  }

  // ICommand 通用实现 — 所有按钮事件绑定的基础
  public class RelayCommand : ICommand
  {
      private readonly Action<object> _execute;
      public RelayCommand(Action<object> e, Func<object, bool> c = null) { _execute = e; _canExecute = c; }
      public bool CanExecute(object p) => _canExecute == null || _canExecute(p);
      public void Execute(object p) => _execute(p);
  }

2. 数据库上下文 (AppDbContext.cs)
------------------------------------
  // EF6 DbContext — 集中管理 18 个数据库实体 + 3 个复合唯一约束
  public class AppDbContext : DbContext
  {
      public DbSet<User> Users { get; set; }
      public DbSet<Club> Clubs { get; set; }
      // ... 共 18 个 DbSet
      protected override void OnModelCreating(DbModelBuilder m)
      {
          m.Entity<ClubMember>().HasIndex(cm => new { cm.ClubId, cm.UserId }).IsUnique();
          m.Entity<ActivityRegistration>().HasIndex(ar => new { ar.ActivityId, ar.UserId }).IsUnique();
          m.Entity<User>().HasIndex(u => u.StudentId).IsUnique();
      }
  }

3. 权限控制中心 (Session.cs)
------------------------------------
  // 全局静态会话 — 存储当前登录用户 + 提供所有权限判断
  public static class Session
  {
      public static User CurrentUser { get; set; }
      public static bool IsAdmin => SystemRole == "系统管理员";
      public static bool IsVisitor => !IsAdmin && (CurrentUser?.ClubMembers?.Count ?? 0) == 0;

      // 身份标签（右上角显示）
      public static string EffectiveRoleLabel
      {
          get {
              if (SystemRole == "系统管理员") return "系统管理员";
              if (CurrentUser?.ClubMembers?.Any(m => m.Role == "社长") == true) return "社长";
              if (CurrentUser?.ClubMembers?.Any(m => m.Role == "干事") == true) return "干事";
              if (CurrentUser?.ClubMembers?.Any() == true) return "普通成员";
              return "游客";
          }
      }

      // 我管辖的社团 ID 列表 (社长或干事)
      public static List<int> MyManagedClubIds =>
          CurrentUser?.ClubMembers?.Where(m => m.Role == "社长" || m.Role == "干事")
          .Select(m => m.ClubId).ToList() ?? new List<int>();
  }

  private void DgFiles_PreviewMouseWheel(object sender, MouseWheelEventArgs e)
  {
      e.Handled = true;
      var sv = FindParent<ScrollViewer>(sender as DependencyObject);
      sv?.ScrollToVerticalOffset(sv.VerticalOffset - e.Delta / 3.0);
  }
